Conectar aos Control Sheets
Publique um Apps Script que leia os control sheets de todos os professores e retorne JSON. Siga os passos abaixo.
1
Abra qualquer planilha → Extensões → Apps Script. Cole o script abaixo e salve com Ctrl+S.
const SHEET_IDS = { 'Alesson': '1oqVLjEERQk9VVvK7G7pz4OlQHfE3NmqOCQf0M9ZUVMI', 'Angie': '1tfPDgsKygQy_cE_RKfjALUBqZtbVTOj-Tbwl_C6vC5c', 'Eliza': '1GgmPfRHaVoS9EiSLt8kvMUBDgDk2upi9LY9sHQEgsNo', 'Flávia': '1Y2m6-2l-DEnZzxJ3iE7v1033tE02jldkjKVwycm6RK8', 'Gabriela': '1XXdbvagMUoCjHjAZmJD9iz6Rh9XtvrV18WEzYWDzUPM', 'Gabes': '1Rhk8pcvTFxtOduGFEgBpA3GtR8V1Lh92kDDFYg1mEY0', 'Juliana': '1EW5niOgx5VAZSqlb46v-UlzRl3ewTjsmIhD5xknA0eE', 'Laura': '1XNHgIKalR7-IDDWia8WpQEmbCUlr-Er2AR4ugvBoBg4', 'Patricia': '1ZNDyNV4JyooZCnKqeloop_GNZGGmzR0ZF011BRZqaCs', 'Samu': '1RIj9z-c0e6NWdSaX2SevN-jDoJruPknoBqTC2J-KKBo' }; function doGet() { const out = { teachers:{}, updated: new Date().toISOString() }; for (const [name, id] of Object.entries(SHEET_IDS)) { try { const ss = SpreadsheetApp.openById(id); out.teachers[name] = parseSheetStudents(ss, name); } catch(e) { out.teachers[name] = []; } } return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); } function parseSheetStudents(ss, teacher) { const students = []; ss.getSheets().forEach(sheet => { const rows = sheet.getDataRange().getValues(); let cur = null; rows.forEach(row => { const r = row.map(c => String(c||'').trim()); // Detecta código VIP/GRU/KID const codeCol = r.findIndex(c => /^(VIP|GRU|KID)\s*-\s*\d+/i.test(c)); if (codeCol >= 0) { if (cur) students.push(cur); // Nome geralmente na mesma linha em outra coluna const nameCol = r.findIndex((c,i) => i !== codeCol && c.length > 1 && !/^\d+$/.test(c) && !/^(VIP|GRU|cod|aluno|prof)/i.test(c) && i > 4); cur = { teacher, code: r[codeCol], name: nameCol>=0 ? r[nameCol] : '', classes: [], stats:{ok:0,nc:0,ca:0,cp:0,f:0,r:0} }; return; } // Linha de aula: col[2] tem status válido e col[1] tem número const st = r[2].toUpperCase(); if (cur && /^\d+$/.test(r[1]) && ['OK','NC','CA','CP','F','R','AB','VC','CF'].includes(st)) { const dateParts = (r[4]||'').split('/'); const month = dateParts.length>=2 ? parseInt(dateParts[1])||0 : 0; cur.classes.push({ status:st, month, date:r[4], time:r[5], content:r[6], obs:r[7], teacher: r[3]||teacher }); if(st==='OK')cur.stats.ok++; else if(st==='NC')cur.stats.nc++; else if(st==='CA')cur.stats.ca++; else if(st==='CP')cur.stats.cp++; else if(st==='F')cur.stats.f++; else if(st==='R')cur.stats.r++; } }); if (cur) students.push(cur); }); return students; }
